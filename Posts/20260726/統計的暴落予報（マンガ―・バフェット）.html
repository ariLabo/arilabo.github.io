def save_to_drive(df_new, filename):
    path = os.path.join(SAVE_PATH, filename)
    key_col = '市場' if '市場' in df_new.columns else '国'
    if os.path.exists(path):
        try:
            df_old = pd.read_csv(path)
            df_combined = pd.concat([df_new, df_old], ignore_index=True)
            df_combined.drop_duplicates(subset=['実行日', key_col], keep='first').to_csv(path, index=False, encoding="utf-8-sig")
            print(f"✅ Google Drive保存完了: {filename}")
        except:
            df_new.to_csv(path, index=False, encoding="utf-8-sig")
    else:
        df_new.to_csv(path, index=False, encoding="utf-8-sig")
        print(f"🆕 新規ファイル作成: {filename}")

def format_jpy(n_billion_jpy):
    if n_billion_jpy <= 0 or pd.isna(n_billion_jpy): return "N/A"
    t, b = int(n_billion_jpy // 1000), int(n_billion_jpy % 1000)
    res = ""
    if t > 0: res += f"{t}兆"
    if b > 0: res += f"{b}億"
    return res + "円" if res != "" else "0円"

def get_fred_raw(series_id):
    if not series_id: return 0.0
    try:
        url = f"https://api.stlouisfed.org/fred/series/observations?series_id={series_id}&api_key={FRED_API_KEY}&file_type=json&sort_order=desc&limit=1"
        res = requests.get(url, timeout=10).json()
        if 'observations' in res and len(res['observations']) > 0:
            val = res['observations'][0]['value']
            return float(val) if val != "." else 0.0
        return 0.0
    except: return 0.0

def normalize_gdp(val, country):
    if val <= 0:
        return {"ドイツ": 4200, "インド": 320000, "ブラジル": 11000, "韓国": 2200000, "ロシア": 190000, "イギリス": 2600}.get(country, 0.0)
    if val > 10**11: return val / 10**6
    if val > 10**9: return val / 10**3
    return val

def assign_general_level(val):
    if pd.isna(val): return 4.0
    if val >= 15: return 1.0
    if val >= 8:  return 2.0
    if val >= 2:  return 3.0
    if val > -2:  return 4.0
    if val > -8:  return 5.0
    return 6.0

def assign_volatility_level(ratio):
    if pd.isna(ratio): return 4.0

    # --- 警戒・危機（両極端） ---
    if ratio >= 1.5: return 1.0   # L1: パニック（急落中）
    if ratio >= 1.25: return 2.0  # L2: 荒れ始め
    if ratio <= 0.70: return 1.0  # L1: 嵐の前の静けさ（異常な無風）
    if ratio <= 0.85: return 2.0  # L2: 静かすぎ（不気味な楽観）

    # --- 割安・適正（健全な波動） ---
    if 0.95 <= ratio <= 1.05: return 6.0  # L6: 理想的な波動（健全な買い場）
    if 0.90 <= ratio <= 1.15: return 5.0  # L5: 適正な押し目

    return 4.0  # L4: 標準

def assign_hyg_tlt_level(val):
    if pd.isna(val): return 4.0
    if val >= 3.0: return 1.0   # 歴史的過熱
    if val >= 2.0: return 2.0   # 過熱警戒
    if val <= -2.0: return 2.0  # 暴落・パニック発生中
    if val >= 1.0: return 3.0   # やや強気
    if val <= -1.1: return 6.0  # 絶好の買い場
    if val <= -0.6: return 5.0  # 割安
    return 4.0

def assign_usdjpy_level(val):
    """USD/JPY専用：急激な円安(L1)と急激な円高(L2)の両面を判定"""
    if pd.isna(val): return 4.0
    if val >= 8.0: return 1.0   # L1: 円安バブル・キャリートレードの極致。
    if val >= 4.0: return 3.0   # L3: 円安進行
    if val <= -6.0: return 2.0  # L2: 急激な円高ショック（パニック）
    if val <= -4.0: return 6.0  # L6: 絶好の買い場（底値）
    if val <= -2.0: return 5.0  # L5: 押し目
    return 4.0                  # L4: 正常

def assign_stock_bond_level(val):
    """株/債券比率専用：株への過度な集中(L1)と株の投げ売り(L2)を判定"""
    if pd.isna(val): return 4.0
    if val >= 15.0: return 1.0   # L1: 株バブル（過熱の極致）
    if val >= 8.0:  return 3.0   # L3: 株偏重
    if val <= -12.0: return 2.0  # L2: 株パニック（債券への資金逃避）
    if val <= -8.0:  return 6.0  # L6: 絶好の株買い場（逆張り好機）
    if val <= -2.0:  return 5.0  # L5: 適正な押し目
    return 4.0                   # L4: 正常レンジ


def assign_audjpy_level(val):
    """AUD/JPY専用：円安バブル(L1)と急激な円高ショック(L2)を判定"""
    if pd.isna(val): return 4.0
    if val >= 8.0: return 1.0   # L1: 円安バブル・楽観の極致
    if val >= 4.0: return 3.0   # L3: やや過熱
    if val <= -8.0: return 2.0  # L2: 円高ショック・パニック発生（警戒）
    if val <= -4.5: return 6.0  # L6: 絶好の買い場（底値圏）
    if val <= -2.0: return 5.0  # L5: 健全な調整（押し目）
    return 4.0                  # L4: 正常

def assign_gold_copper_level(val):
    """金/銅比率：不況(高)と過熱(低)の両方を判定"""
    if pd.isna(val): return 4.0
    if val >= 2.0: return 1.0   # L1: 強烈な不況懸念
    if val >= 1.5: return 2.0   # L2: 景気後退の兆候
    if val <= -2.0: return 1.0  # L1: 資源バブル・利上げ直前警戒
    if val <= -1.5: return 2.0  # L2: 景気過熱
    if -0.5 <= val <= 0.5: return 6.0  # L6: 理想的な成長バランス
    return 4.0

def assign_btc_level(val):
    if pd.isna(val): return 4.0
    if val >= 60: return 1.0
    if val >= 30: return 2.0
    if val >= 10: return 3.0
    if val > -10: return 4.0
    if val > -30: return 5.0
    return 6.0

def assign_buffett_level(ratio_percent):
    if pd.isna(ratio_percent) or ratio_percent <= 0: return 4
    if ratio_percent >= 150: return 1
    if ratio_percent >= 120: return 2
    if ratio_percent >= 100: return 3
    if ratio_percent >= 75:  return 4
    if ratio_percent >= 50:  return 5
    return 6

# ==== 4. メイン分析ロジック ====
def run_analysis():
    today = datetime.datetime.today().strftime("%Y-%m-%d")
    print(f"🚀 {today} のデータを取得・分析中...")

    fx_list = ["USDJPY=X", "EURUSD=X", "GBPUSD=X", "USDCNY=X", "USDINR=X", "USDBRL=X", "USDKRW=X", "USDRUB=X", "USDCAD=X", "AUDJPY=X"]
    idx_tickers = [cfg['index'] for cfg in COUNTRY_CONFIG.values()]
    com_tickers = ["GC=F", "HG=F", "BTC-USD", "CL=F", "TLT", "HYG", "^GSPC", "^N225"]

    all_tickers = list(set(fx_list + idx_tickers + com_tickers))
    df_yf = yf.download(all_tickers, period="21y", progress=False, auto_adjust=True)["Close"].ffill()
    latest = df_yf.iloc[-1]
    usdjpy = latest["USDJPY=X"]

    # --- 共通指標計算 ---
    gc_ratio = (df_yf['GC=F'] / df_yf['HG=F']).dropna()
    ht_ratio = (df_yf['HYG'] / df_yf['TLT']).dropna()
    common_metrics = {
        "金/銅比率 Z-score": (gc_ratio.iloc[-1] - gc_ratio.mean()) / gc_ratio.std() * 5,
        "原油価格 200日MA乖離率": ((latest['CL=F'] / df_yf['CL=F'].rolling(200).mean().iloc[-1]) - 1) * 100,
        "BTC 200日MA乖離率": ((latest['BTC-USD'] / df_yf['BTC-USD'].rolling(200).mean().iloc[-1]) - 1) * 100,
        "USD/JPY 60日モメンタム(%)": ((latest['USDJPY=X'] / df_yf['USDJPY=X'].rolling(60).mean().iloc[-1]) - 1) * 100,
        "AUD/JPY 60日モメンタム(%)": ((latest['AUDJPY=X'] / df_yf['AUDJPY=X'].rolling(60).mean().iloc[-1]) - 1) * 100,
        "HYG/TLT乖離率": ((ht_ratio.rolling(20).mean().iloc[-1] / ht_ratio.rolling(200).mean().iloc[-1]) - 1) * 100
    }

    # --- マンガースコア分析 ---
    m_rows = []
    for label, tkr in [("日本", "^N225"), ("米国", "^GSPC")]:
        p = df_yf[tkr]
        v_ratio = p.pct_change().rolling(20).std().iloc[-1] / p.pct_change().rolling(200).std().iloc[-1]
        spec = {
            "ボラティリティ比率(VI代用)": v_ratio,
            "超長期評価乖離率(20年MA)": ((latest[tkr] / p.rolling(5000).mean().iloc[-1]) - 1) * 100,
            "MA400乖離率": ((latest[tkr] / p.rolling(400).mean().iloc[-1]) - 1) * 100,
            "株/債券比率": (((latest[tkr]/latest['TLT']) / (df_yf[tkr]/df_yf['TLT']).rolling(200).mean().iloc[-1]) - 1) * 100
        }
        w_sum_raw, row = 0, {"実行日": today, "市場": label}
        for k, w in M_WEIGHTS.items():
            val = spec.get(k, common_metrics.get(k))
            # 修正後のループ内分岐
            if k == "ボラティリティ比率(VI代用)":
                lvl = assign_volatility_level(val)
            elif k == "HYG/TLT乖離率":
                lvl = assign_hyg_tlt_level(val) # ここを追加
            elif k == "株/債券比率":
                lvl = assign_stock_bond_level(val) # ここを追加
            elif k == "USD/JPY 60日モメンタム(%)":
                lvl = assign_usdjpy_level(val) # ここを追加
            elif k == "AUD/JPY 60日モメンタム(%)":
                lvl = assign_audjpy_level(val) # ここを追加
            elif k == "BTC 200日MA乖離率":
                lvl = assign_btc_level(val)
            else:
                lvl = assign_general_level(val)

            w_sum_raw += lvl * w
            row[k] = f"{val:.2f} (L{int(lvl)})"

        total_36 = w_sum_raw * 6
        if total_36 <= 8.5: crash_lvl = 1
        elif total_36 <= 14.5: crash_lvl = 2
        elif total_36 <= 20.5: crash_lvl = 3
        elif total_36 <= 26.5: crash_lvl = 4
        elif total_36 <= 32.5: crash_lvl = 5
        else: crash_lvl = 6
        row["暴落レベル"], row["総合点(Max36)"] = crash_lvl, f"{total_36:.2f}"
        m_rows.append(row)

    m_df = pd.DataFrame(m_rows)[MUNGER_COL_ORDER]

    # --- バフェット指数分析 ---
    b_rows = []
    for country, cfg in COUNTRY_CONFIG.items():
        idx_val = latest[cfg['index']]
        if country == "日本":
            fx_jpy, fx_display = 1.0, "1.00 円"
        elif country == "アメリカ":
            fx_jpy, fx_display = usdjpy, f"{usdjpy:.2f} 円/ドル"
        else:
            rate = latest[cfg['usd_pair']]
            fx_jpy = usdjpy / rate if "USD" == cfg['usd_pair'][:3] else rate * usdjpy
            fx_display = f"{fx_jpy:.2f} 円/{cfg['unit']}"

        mcap_jpy = (idx_val * cfg['factor']) * fx_jpy
        display_country = country
        if country in ["中国", "カナダ"]:
            g_local = cfg['fixed_gdp']
            display_country = f"{country}(GDP固定値)"
        else:
            raw_gdp = get_fred_raw(cfg['gdp_sid'])
            g_local = normalize_gdp(raw_gdp, country)

        if g_local > 0:
            gdp_jpy = g_local * fx_jpy
            ratio = (mcap_jpy / gdp_jpy) * 100
            gdp_ja = format_jpy(gdp_jpy)
            ratio_str = f"{ratio:.2f}%"
            ratio_num = ratio
        else:
            gdp_jpy, gdp_ja, ratio_str, ratio_num = 0, "取得失敗", "N/A", -1

        b_lvl = assign_buffett_level(ratio_num)
        b_rows.append({
            "実行日": today, "国": display_country, "暴落レベル": b_lvl,
            "バフェット指数(%)": ratio_str, "GDP(円換算)": gdp_ja,
            "時価総額近似(円換算)": format_jpy(mcap_jpy),
            "株価指数生データ": f"{idx_val:,.0f} {cfg['unit']}",
            "対円為替": fx_display, "_ratio_sort": ratio_num
        })

    b_df = pd.DataFrame(b_rows).sort_values("_ratio_sort", ascending=False)
    b_df = b_df[["実行日", "国", "暴落レベル", "バフェット指数(%)", "GDP(円換算)", "時価総額近似(円換算)", "株価指数生データ", "対円為替"]]

    save_to_drive(m_df, "munger_score.csv")
    save_to_drive(b_df, "buffett_index.csv")

    return m_df, b_df
